func ref(t *testing.T, rel string) string {
	t.Helper()
	b, err := os.ReadFile(rel)
	if err != nil {
		t.Fatalf("reference source missing: %v", err)
	}
	return string(b)
}

func must1(t *testing.T, re, text, what string) string {
	t.Helper()
	m := regexp.MustCompile(re).FindStringSubmatch(text)
	if m == nil {
		t.Fatalf("source does not contain %s (pattern %q)", what, re)
	}
	return m[1]
}

func hexes(t *testing.T, text string) [][]byte {
	t.Helper()
	var out [][]byte
	for _, m := range regexp.MustCompile(`(?:hd|dh)\("([0-9a-f]*)"`).FindAllStringSubmatch(text, -1) {
		b, err := hex.DecodeString(m[1])
		if err != nil {
			t.Fatal(err)
		}
		out = append(out, b)
	}
	return out
}

func funcBody(t *testing.T, src, name string) string {
	t.Helper()
	i := strings.Index(src, "func "+name+"(")
	if i < 0 {
		t.Fatalf("source has no func %s", name)
	}
	rest := src[i:]
	j := strings.Index(rest, "\n}\n")
	if j < 0 {
		t.Fatalf("func %s not terminated", name)
	}
	return rest[:j]
}

// --- DSSE v1 ---

func TestC14a_DSSEPAEExampleFromSpec(t *testing.T) {
	spec := ref(t, refDSSE)
	block := func(label string) string {
		return must1(t, "(?s)"+label+":\\s*```none\\n(.*?)\\n```", spec, "block "+label)
	}
	body, ptype, want := block("SERIALIZED_BODY"), block("PAYLOAD_TYPE"), block("PAE")
	if got := string(attest.PAE(ptype, []byte(body))); got != want {
		t.Fatalf("PAE(%q, %q) = %q, spec says %q", ptype, body, got, want)
	}
	// способность: изменённый вход даёт другой результат
	if string(attest.PAE(ptype+"x", []byte(body))) == want {
		t.Fatal("PAE ignores the payload type")
	}
}

// --- OpenSSL и C2SP signed-note ---

func TestC14b_OpenSSLVerifiesOurSignatures(t *testing.T) {
	if _, err := exec.LookPath("openssl"); err != nil {
		t.Fatal("openssl not found: control cannot run")
	}
	note := ref(t, refNote)
	for _, must := range []string{"dash (U+2014)", "The first four bytes in the", "key ID = SHA-256(key name || 0x0A || 0x01 || 32-byte Ed25519 public key)[:4]"} {
		if !strings.Contains(note, must) {
			t.Fatalf("signed-note source lacks %q", must)
		}
	}
	dir := t.TempDir()
	s := newSigner(t)
	pubPath := filepath.Join(dir, "pub.pem")
	if err := keys.WritePublic(pubPath, s.Public()); err != nil {
		t.Fatal(err)
	}
	if out, err := exec.Command("openssl", "pkey", "-pubin", "-in", pubPath, "-noout", "-text").CombinedOutput(); err != nil || !strings.Contains(string(out), "ED25519") {
		t.Fatalf("openssl cannot read our public key as ED25519: %v %s", err, out)
	}
	verify := func(msg, sig []byte) error {
		mp, sp := filepath.Join(dir, "msg"), filepath.Join(dir, "sig")
		_ = os.WriteFile(mp, msg, 0o600)
		_ = os.WriteFile(sp, sig, 0o600)
		return exec.Command("openssl", "pkeyutl", "-verify", "-pubin", "-inkey", pubPath, "-rawin", "-in", mp, "-sigfile", sp).Run()
	}

	st, err := attest.StatementFromEvent(testEvent(t, 1))
	if err != nil {
		t.Fatal(err)
	}
	env, err := attest.SignStatement(st, s)
	if err != nil {
		t.Fatal(err)
	}
	payload, _ := base64.StdEncoding.DecodeString(env.Payload)
	sig, _ := base64.StdEncoding.DecodeString(env.Signatures[0].Sig)
	pae := attest.PAE(env.PayloadType, payload)
	if err := verify(pae, sig); err != nil {
		t.Fatalf("OpenSSL rejected our DSSE signature: %v", err)
	}
	if err := verify(append(append([]byte{}, pae...), ' '), sig); err == nil {
		t.Fatal("(capability) OpenSSL accepted a corrupted message")
	}

	signed, err := tlog.SignCheckpoint(tlog.Checkpoint{Origin: base, Size: 5, Root: tlog.EmptyRoot()}, s)
	if err != nil {
		t.Fatal(err)
	}
	i := strings.Index(string(signed), "\n\n— ")
	text, line := signed[:i+1], strings.TrimSpace(string(signed[i+2:]))
	f := strings.SplitN(strings.TrimPrefix(line, "— "), " ", 2)
	if len(f) != 2 || f[0] != base {
		t.Fatalf("unexpected signature line %q", line)
	}
	blob, _ := base64.StdEncoding.DecodeString(f[1])
	if len(blob) != 4+ed25519.SignatureSize {
		t.Fatalf("signature blob must be 4+64 bytes, got %d", len(blob))
	}
	// key ID по формуле из спецификации, независимо от кода репозитория
	h := sha256.New()
	h.Write([]byte(base))
	h.Write([]byte{0x0a, 0x01})
	h.Write(s.Public())
	if got, want := hex.EncodeToString(blob[:4]), hex.EncodeToString(h.Sum(nil)[:4]); got != want {
		t.Fatalf("key ID in our note %s, by the spec formula %s", got, want)
	}
	if err := verify(text, blob[4:]); err != nil {
		t.Fatalf("OpenSSL rejected our checkpoint-note signature: %v", err)
	}
	if err := verify(append([]byte("x"), text...), blob[4:]); err == nil {
		t.Fatal("(capability) OpenSSL accepted a corrupted note")
	}
}

func TestC14c_VerifierKeyFromReferenceImplementation(t *testing.T) {
	out, err := exec.Command("go", "env", "GOMODCACHE").Output()
	if err != nil {
		t.Fatal(err)
	}
	b, err := os.ReadFile(filepath.Join(strings.TrimSpace(string(out)), refXNote))
	if err != nil {
		t.Fatalf("эталон без источника: модуль golang.org/x/mod@v0.22.0 не загружен (запустите c13_interop.sh): %v", err)
	}
	vkey := must1(t, `vkey := "(PeterNeumann\+[^"]+)"`, string(b), "vkey in note.go")
	parts := strings.SplitN(vkey, "+", 3)
	raw, err := base64.StdEncoding.DecodeString(parts[2])
	if err != nil || len(raw) != 33 || raw[0] != 1 {
		t.Fatalf("source vkey malformed: %v len=%d", err, len(raw))
	}
	if got := tlog.VerifierKey(parts[0], ed25519.PublicKey(raw[1:])); got != vkey {
		t.Fatalf("our vkey %s differs from the reference implementation's %s", got, vkey)
	}
}

// --- RFC 6962 / векторы Certificate Transparency (transparency-dev/merkle v0.0.2) ---

func TestC14e_CTRootsFromReferenceSource(t *testing.T) {
	src := ref(t, refConsts)
	leaves := hexes(t, funcBody(t, src, "LeafInputs"))
	roots := hexes(t, funcBody(t, src, "RootHashes")) // индексы 1…8; 0-й в источнике — EmptyRootHash()
	empty := hexes(t, funcBody(t, src, "EmptyRootHash"))
	if len(leaves) != 8 || len(roots) != 8 || len(empty) != 1 {
		t.Fatalf("source parse: leaves=%d roots=%d empty=%d", len(leaves), len(roots), len(empty))
	}
	if r := tlog.EmptyRoot(); hex.EncodeToString(r[:]) != hex.EncodeToString(empty[0]) {
		t.Fatalf("empty root differs from the source")
	}
	var lh []tlog.Hash
	for _, l := range leaves {
		lh = append(lh, tlog.LeafHash(l))
	}
	for n := 1; n <= 8; n++ {
		r := tlog.RootFromLeaves(lh[:n])
		if hex.EncodeToString(r[:]) != hex.EncodeToString(roots[n-1]) {
			t.Fatalf("root of %d leaves = %x, source says %x", n, r, roots[n-1])
		}
	}
}

func TestC14f_CTInclusionProofsFromReferenceSource(t *testing.T) {
	cs := ref(t, refConsts)
	leaves := hexes(t, funcBody(t, cs, "LeafInputs"))
	roots := hexes(t, funcBody(t, cs, "RootHashes"))
	var lh []tlog.Hash
	for _, l := range leaves {
		lh = append(lh, tlog.LeafHash(l))
	}
	ps := ref(t, refProofs)
	i, j := strings.Index(ps, "inclusionProofs = []inclusionProofTestVector{"), strings.Index(ps, "consistencyProofs = ")
	if i < 0 || j < i {
		t.Fatal("source: inclusionProofs block not found")
	}
	block := ps[i:j]
	starts := regexp.MustCompile(`(?m)^\t\t\{(\d+), (\d+), `).FindAllStringSubmatchIndex(block, -1)
	checked := 0
	for k, m := range starts {
		end := len(block)
		if k+1 < len(starts) {
			end = starts[k+1][0]
		}
		leaf, _ := strconv.Atoi(block[m[2]:m[3]])
		size, _ := strconv.Atoi(block[m[4]:m[5]])
		if leaf == 0 || size == 0 {
			continue // в источнике нумерация листьев с 1; нулевые векторы — граничные
		}
		var want []tlog.Hash
		for _, h := range hexes(t, block[m[0]:end]) {
			var x tlog.Hash
			copy(x[:], h)
			want = append(want, x)
		}
		idx := uint64(leaf - 1)
		got, err := tlog.InclusionProof(lh[:size], idx)
		if err != nil || len(got) != len(want) {
			t.Fatalf("vector (%d,%d): err=%v len=%d want %d", leaf, size, err, len(got), len(want))
		}
		for x := range want {
			if got[x] != want[x] {
				t.Fatalf("vector (%d,%d): path element %d differs from the source", leaf, size, x)
			}
		}
		var root tlog.Hash
		copy(root[:], roots[size-1])
		if err := tlog.VerifyInclusion(lh[idx], idx, uint64(size), want, root); err != nil {
			t.Fatalf("vector (%d,%d): our verifier rejects the source's proof: %v", leaf, size, err)
		}
		checked++
	}
	if checked != 5 {
		t.Fatalf("expected 5 non-trivial vectors in the source, checked %d", checked)
	}
}

func TestC14g_InTotoStringsFromSpecs(t *testing.T) {
	stmt := ref(t, refStmt)
	if got := must1(t, `"_type": "([^"]+)"`, stmt, "_type in statement.md"); got != attest.StatementType {
		t.Fatalf("StatementType %q, spec %q", attest.StatementType, got)
	}
	if got := must1(t, "`payloadType` MUST be set to `([^`]+)`", ref(t, refEnv), "payloadType in envelope.md"); got != attest.PayloadType {
		t.Fatalf("PayloadType %q, spec %q", attest.PayloadType, got)
	}
	tr := ref(t, refTR)
	if got := must1(t, `Type URI: (\S+)`, tr, "Type URI in test-result.md"); got != event.PredicateTestResult {
		t.Fatalf("PredicateTestResult %q, spec %q", event.PredicateTestResult, got)
	}
	for _, v := range []string{event.ResultPassed, event.ResultWarned, event.ResultFailed} {
		if !strings.Contains(tr, v) {
			t.Fatalf("test-result spec does not mention %s", v)
		}
	}
	st, _ := attest.StatementFromEvent(testEvent(t, 1))
	b, _ := json.Marshal(st)
	for _, k := range []string{`"_type"`, `"subject"`, `"predicateType"`, `"predicate"`} {
		if !strings.Contains(stmt, k) || !strings.Contains(string(b), k) {
			t.Fatalf("field %s missing in the spec or in our Statement JSON", k)
		}
	}
}

// --- дополнительный тест без источника (в основание не берётся) ---

func hashLeaf(d []byte) [32]byte { return sha256.Sum256(append([]byte{0}, d...)) }
func hashNode(l, r [32]byte) [32]byte {
	return sha256.Sum256(append(append([]byte{1}, l[:]...), r[:]...))
}
func lpow2(n int) int {
	k := 1
	for k*2 < n {
		k *= 2
	}
	return k
}
func mth(d [][]byte) [32]byte {
	if len(d) == 0 {
		return sha256.Sum256(nil)
	}
	if len(d) == 1 {
		return hashLeaf(d[0])
	}
	k := lpow2(len(d))
	return hashNode(mth(d[:k]), mth(d[k:]))
}
func pathOf(m int, d [][]byte) [][32]byte {
	if len(d) <= 1 {
		return nil
	}
	k := lpow2(len(d))
	if m < k {
		return append(pathOf(m, d[:k]), mth(d[k:]))
	}
	return append(pathOf(m-k, d[k:]), mth(d[:k]))
}

func TestC14s_IndependentRFC6962(t *testing.T) {
	for n := 1; n <= 70; n++ {
		var data [][]byte
		var leaves []tlog.Hash
		for i := 0; i < n; i++ {
			b := []byte{byte(i), byte(n), 7}
			data = append(data, b)
			leaves = append(leaves, tlog.LeafHash(b))
		}
		root := tlog.RootFromLeaves(leaves)
		if root != tlog.Hash(mth(data)) {
			t.Fatalf("n=%d: root differs from the second implementation", n)
		}
		for m := 0; m < n; m++ {
			ours, err := tlog.InclusionProof(leaves, uint64(m))
			if err != nil {
				t.Fatal(err)
			}
			ref := pathOf(m, data)
			if len(ours) != len(ref) {
				t.Fatalf("n=%d m=%d: proof length %d != %d", n, m, len(ours), len(ref))
			}
			for j := range ref {
				if ours[j] != tlog.Hash(ref[j]) {
					t.Fatalf("n=%d m=%d: proof element %d differs", n, m, j)
				}
			}
		}
	}
}
